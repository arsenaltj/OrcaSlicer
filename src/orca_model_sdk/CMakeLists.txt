cmake_minimum_required(VERSION 3.22)
project(orca_model_sdk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ORCA_MODEL_BUILD_TESTS "Build SDK-only consumers" ON)
option(ORCA_MODEL_REGION_CORE "Compile existing native region algorithm independently" OFF)
option(ORCA_MODEL_REPAIR_CORE "Compile the existing desktop pure repair core" OFF)
option(ORCA_MODEL_GLB_CGAL "Enable strict CGAL predicates for original GLB safety" OFF)
set(ORCA_MODEL_CGAL_INCLUDE "" CACHE PATH "Pinned CGAL 6.2.1 include root")
set(ORCA_MODEL_BOOST_ROOT "" CACHE PATH "Pinned Boost 1.84 source root")
set(ORCA_MODEL_SOURCE_COMMIT "uncommitted" CACHE STRING "Source provenance baseline")
if(ORCA_MODEL_SOURCE_COMMIT STREQUAL "uncommitted")
 execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}/../.." rev-parse HEAD
  OUTPUT_VARIABLE ORCA_MODEL_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
endif()
add_library(orca_model_sdk SHARED model_sdk.cpp runtime.cpp model_io.cpp model_open.cpp digest.cpp document.cpp scene_file.cpp operations.cpp manufacturing_bridge.cpp glb_safety.cpp ../orca_mobile_slicing_sdk/platform/AndroidSha256Digest.cpp)
target_compile_definitions(orca_model_sdk PRIVATE ORCA_MODEL_BUILD ORCA_MODEL_SOURCE_COMMIT="${ORCA_MODEL_SOURCE_COMMIT}")
target_include_directories(orca_model_sdk PUBLIC include PRIVATE ../../deps_src .. ../orca_mobile_slicing_sdk/include)
if(ORCA_MODEL_GLB_CGAL)
 if(NOT EXISTS "${ORCA_MODEL_CGAL_INCLUDE}/CGAL/version.h" OR NOT EXISTS "${ORCA_MODEL_BOOST_ROOT}/libs/config/include/boost/version.hpp")
  message(FATAL_ERROR "GLB predicates require explicit pinned CGAL and Boost source headers")
 endif()
 file(GLOB glb_boost_headers LIST_DIRECTORIES true "${ORCA_MODEL_BOOST_ROOT}/libs/*/include")
 file(GLOB glb_boost_numeric_headers LIST_DIRECTORIES true "${ORCA_MODEL_BOOST_ROOT}/libs/numeric/*/include")
 target_sources(orca_model_sdk PRIVATE glb_safety_cgal.cpp)
 target_include_directories(orca_model_sdk SYSTEM PRIVATE ${ORCA_MODEL_CGAL_INCLUDE} ${glb_boost_headers} ${glb_boost_numeric_headers})
 target_compile_definitions(orca_model_sdk PRIVATE ORCA_MODEL_HAS_GLB_CGAL CGAL_DISABLE_GMP=1 CGAL_NO_CORE=1)
endif()
set_target_properties(orca_model_sdk PROPERTIES OUTPUT_NAME orca_model_sdk_c CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(ORCA_MODEL_REGION_CORE)
 include(cmake/RegionCore.cmake)
endif()
if(ANDROID)
 target_link_options(orca_model_sdk PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
 target_link_libraries(orca_model_sdk PRIVATE dl)
endif()
add_executable(orca_model_runner runner.cpp)
target_include_directories(orca_model_runner PRIVATE ../../deps_src)
target_link_libraries(orca_model_runner PRIVATE orca_model_sdk)
if(ORCA_MODEL_BUILD_TESTS)
 enable_testing()
 add_executable(orca_model_consumer tests/consumer.c)
 target_link_libraries(orca_model_consumer PRIVATE orca_model_sdk)
 add_test(NAME public_c_consumer COMMAND orca_model_consumer)
 if(ORCA_MODEL_GLB_CGAL)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME glb_safety_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/glb_safety_test.py $<TARGET_FILE:orca_model_sdk>)
  if(ORCA_MODEL_REPAIR_CORE)
   add_test(NAME glb_scene_profile_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/glb_scene_profile_test.py $<TARGET_FILE:orca_model_sdk>)
  endif()
 endif()
endif()
install(TARGETS orca_model_sdk orca_model_runner RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)

if(ORCA_MODEL_REPAIR_CORE)
 include(cmake/RepairCore.cmake)
endif()
if(ORCA_MODEL_BUILD_TESTS AND NOT ANDROID)
 add_executable(orca_android_publish_policy_test tests/android_publish_test.cpp)
 if(MSVC)
  target_compile_options(orca_android_publish_policy_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(orca_android_publish_policy_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME android_exclusive_publish_policy COMMAND orca_android_publish_policy_test)
 get_target_property(io_sources orca_model_sdk SOURCES)
 get_target_property(io_includes orca_model_sdk INCLUDE_DIRECTORIES)
 get_target_property(io_definitions orca_model_sdk COMPILE_DEFINITIONS)
 get_target_property(io_links orca_model_sdk LINK_LIBRARIES)
 add_library(orca_model_io_test SHARED ${io_sources})
 target_include_directories(orca_model_io_test PRIVATE ${io_includes})
 target_compile_definitions(orca_model_io_test PRIVATE ${io_definitions} ORCA_MODEL_IO_TESTING)
 # LINK_LIBRARIES is unset in the default SDK-only profile. CMake returns
 # io_links-NOTFOUND in that case; only propagate actual optional dependencies.
 if(io_links)
  target_link_libraries(orca_model_io_test PRIVATE ${io_links})
 endif()
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME io_diagnostics_native COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/io_diagnostic_test.py $<TARGET_FILE:orca_model_io_test> $<TARGET_FILE:orca_model_sdk> $<IF:$<BOOL:${ORCA_MODEL_REPAIR_CORE}>,ON,OFF>)
endif()

cmake_minimum_required(VERSION 3.22)
project(orca_model_sdk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ORCA_MODEL_BUILD_TESTS "Build SDK-only consumers" ON)
option(ORCA_MODEL_REGION_CORE "Compile existing native region algorithm independently" OFF)
set(ORCA_MODEL_SOURCE_COMMIT "uncommitted" CACHE STRING "Source provenance baseline")
if(ORCA_MODEL_SOURCE_COMMIT STREQUAL "uncommitted")
 execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}/../.." rev-parse HEAD
  OUTPUT_VARIABLE ORCA_MODEL_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
endif()
add_library(orca_model_sdk SHARED model_sdk.cpp runtime.cpp model_io.cpp model_open.cpp digest.cpp document.cpp scene_file.cpp operations.cpp manufacturing_bridge.cpp ../orca_mobile_slicing_sdk/platform/AndroidSha256Digest.cpp)
target_compile_definitions(orca_model_sdk PRIVATE ORCA_MODEL_BUILD ORCA_MODEL_SOURCE_COMMIT="${ORCA_MODEL_SOURCE_COMMIT}")
target_include_directories(orca_model_sdk PUBLIC include PRIVATE ../../deps_src .. ../orca_mobile_slicing_sdk/include)
set_target_properties(orca_model_sdk PROPERTIES OUTPUT_NAME orca_model_sdk_c CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(ORCA_MODEL_REGION_CORE)
 include(cmake/RegionCore.cmake)
endif()
if(ANDROID)
 target_link_options(orca_model_sdk PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
 target_link_libraries(orca_model_sdk PRIVATE dl)
endif()
add_executable(orca_model_runner runner.cpp)
target_include_directories(orca_model_runner PRIVATE ../../deps_src)
target_link_libraries(orca_model_runner PRIVATE orca_model_sdk)
if(ORCA_MODEL_BUILD_TESTS)
 enable_testing()
 add_executable(orca_model_consumer tests/consumer.c)
 target_link_libraries(orca_model_consumer PRIVATE orca_model_sdk)
 add_test(NAME public_c_consumer COMMAND orca_model_consumer)
endif()
install(TARGETS orca_model_sdk orca_model_runner RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
